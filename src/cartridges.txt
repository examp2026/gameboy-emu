MapperType = {
  NONE, MBC1, MBC2, MBC3,
  MBC5, MBC6, MBC7, MMM01,
  HuC1, HuC3, TAMA5, CAMERA
}

Features = {
  ram :    Bool
  battery: Bool
  timer:   Bool
  rumble:  Bool
  sensor:  Bool
}

parse_type(byte) -> (MapperType, Features)

parse_type(0x00) = (NONE, {ram=F, battery=F, timer=F, rumble=F, sensor=F})
